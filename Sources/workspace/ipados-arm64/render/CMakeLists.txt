function(render_library name type)
 add_library(${name} ${type} ${VPC_${name}_FILES})
 target_link_libraries(${name} PUBLIC source_platform PRIVATE tier0 tier1 mathlib vstdlib interfaces iconv)
 target_compile_definitions(${name} PRIVATE ALLOW_TEXT_MODE=1 VERSION_SAFE_STEAM_API_INTERFACES CSTRIKE_REL_BUILD=1 SURVIVAL_MODE_ENABLED=1)
 target_include_directories(${name} PRIVATE "${SOURCE_ROOT}" "${SOURCE_ROOT}/${name}" "${SOURCE_ROOT}/public/tier2" "${SOURCE_ROOT}/public/tier3" "${METAL_ROOT}")
 if(type STREQUAL "SHARED")
  target_sources(${name} PRIVATE "${SOURCE_ROOT}/public/tier0/memoverride.cpp" "${SOURCE_ROOT}/public/tier0/crtoverride.cpp")
  target_compile_definitions(${name} PRIVATE MEMOVERRIDE_MODULE=${name})
  string(REPLACE "_" "-" ident "${name}")
  set_target_properties(${name} PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER "local.csgo.module.${ident}")
 endif()
endfunction()

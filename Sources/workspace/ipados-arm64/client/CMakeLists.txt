cmake_minimum_required(VERSION 3.31)
project(csgo_ios_client LANGUAGES C CXX OBJC OBJCXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
 message(FATAL_ERROR "Physical iPhoneOS arm64 required")
endif()
set(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(W "${PORT_ROOT}/..")
set(I4_CACHE "$ENV{HOME}/Library/Caches/csgo-ipados-arm64/i4")
set(SOURCE_ROOT "${I4_CACHE}/source")
set(METAL_ROOT "${I4_CACHE}/metal")
set(I1_CACHE "$ENV{HOME}/Library/Caches/csgo-ipados-arm64/i1")
set(I1_BUILD "$ENV{HOME}/Library/Caches/csgo-ipados-arm64/i1/engine-build")
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH "@loader_path/..")
set(CMAKE_INSTALL_NAME_DIR "@rpath")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Frameworks")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(OPENSSL_ROOT_DIR "${I4_CACHE}/../i3/deps")
set(OPENSSL_USE_STATIC_LIBS ON)
set(OPENSSL_INCLUDE_DIR "${I4_CACHE}/../i3/deps/include" CACHE PATH "iOS OpenSSL")
set(OPENSSL_CRYPTO_LIBRARY "${I4_CACHE}/../i3/deps/lib/libcrypto.a" CACHE FILEPATH "iOS OpenSSL")
set(OPENSSL_SSL_LIBRARY "${I4_CACHE}/../i3/deps/lib/libssl.a" CACHE FILEPATH "iOS OpenSSL")
set(CURL_DIR "${I4_CACHE}/../i3/deps/lib/cmake/CURL")
find_package(CURL REQUIRED CONFIG PATHS "${CURL_DIR}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${I4_CACHE}/sdl" "${CMAKE_BINARY_DIR}/sdl")
set_target_properties(SDL2 PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER local.csgo.module.sdl2)
set(MOJO "${W}/macos-arm64/vendor/m6/mojoshader")
add_library(m6_mojoshader STATIC "${MOJO}/mojoshader.c" "${MOJO}/mojoshader_common.c" "${MOJO}/profiles/mojoshader_profile_common.c" "${MOJO}/profiles/mojoshader_profile_metal.c")
target_include_directories(m6_mojoshader PUBLIC "${MOJO}")
target_compile_definitions(m6_mojoshader PRIVATE MOJOSHADER_NO_VERSION_INCLUDE SUPPORT_PROFILE_METAL=1
 SUPPORT_PROFILE_D3D=0 SUPPORT_PROFILE_BYTECODE=0 SUPPORT_PROFILE_HLSL=0 SUPPORT_PROFILE_GLSL=0 SUPPORT_PROFILE_GLSL120=0
 SUPPORT_PROFILE_GLSLES=0 SUPPORT_PROFILE_GLSLES3=0 SUPPORT_PROFILE_ARB1=0 SUPPORT_PROFILE_ARB1_NV=0
 SUPPORT_PROFILE_SPIRV=0 SUPPORT_PROFILE_GLSPIRV=0)
add_library(source_platform INTERFACE)
target_compile_definitions(source_platform INTERFACE GNUC POSIX _POSIX OSX _OSX COMPILER_GCC SOURCE_IOS SOURCE_NATIVE_METAL USE_SDL PANORAMA_ENABLE
 RAD_TELEMETRY_DISABLED CSTRIKE15 _DLL_EXT=.dylib DX_TO_GL_ABSTRACTION GL_GLEXT_PROTOTYPES NO_STEAM NO_STEAM_GAMECOORDINATOR
 SSE2NEON_PRECISE_MINMAX=1 SSE2NEON_PRECISE_DIV=1 SSE2NEON_PRECISE_SQRT=0)
target_include_directories(source_platform INTERFACE "$ENV{HOME}/Library/Caches/csgo-ipados-arm64/i2/headers" "${PORT_ROOT}/render/compat" "${METAL_ROOT}" "${PORT_ROOT}/vendor/SDL2-2.32.10/include" "${SOURCE_ROOT}/public" "${SOURCE_ROOT}/common"
 "${SOURCE_ROOT}/public/tier0" "${SOURCE_ROOT}/public/tier1" "${SOURCE_ROOT}/thirdparty/sse2neon")
target_compile_options(source_platform INTERFACE -fvisibility=hidden -fno-strict-aliasing -Wno-register
 -Wno-invalid-source-encoding -Wno-c++11-narrowing -Wno-deprecated-declarations -Wno-inconsistent-missing-override)
foreach(module tier0 vstdlib vphysics)
 add_library(${module} SHARED IMPORTED)
 set_target_properties(${module} PROPERTIES IMPORTED_LOCATION "${I1_BUILD}/Frameworks/${module}.framework/${module}" FRAMEWORK TRUE)
endforeach()
foreach(module mathlib tier2)
 add_library(${module} STATIC IMPORTED)
 set_target_properties(${module} PROPERTIES IMPORTED_LOCATION "${I1_BUILD}/lib${module}.a")
endforeach()
add_library(interfaces STATIC "${SOURCE_ROOT}/interfaces/interfaces.cpp")
target_link_libraries(interfaces PRIVATE source_platform)
# Rebuild tier1/filesystem from the I2 loading branch, with no graphics linkage.
include("${PORT_ROOT}/render/foundation-render.cmake")
add_library(steam_api STATIC "${PORT_ROOT}/server/compat/steam_unavailable.cpp")
target_link_libraries(steam_api PRIVATE source_platform)
set(BUILD_M4 ON)
set(BUILD_M5 ON)
set(BUILD_M6 ON)
include(cmake/headless.cmake)
list(REMOVE_ITEM VPC_inputsystem_FILES "${SOURCE_ROOT}/inputsystem/joystick_osx.cpp")
list(APPEND VPC_inputsystem_FILES "${SOURCE_ROOT}/inputsystem/joystick_linux.cpp")
include(cmake/client.cmake)
include(cmake/panorama.cmake)
get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(module IN LISTS targets)
 get_target_property(type ${module} TYPE)
 if(type STREQUAL "SHARED_LIBRARY")
  string(REPLACE "_" "-" ident "${module}")
  set_target_properties(${module} PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER "local.csgo.module.${ident}")
 endif()
endforeach()
target_sources(server PRIVATE "${PORT_ROOT}/server/diagnostics.cpp")
target_sources(engine PRIVATE "${PORT_ROOT}/client/host_commands.cpp")
# Preserve M3's original server telemetry types and nullable API contexts. Only
# engine authentication and game coordinator use their complete NO_STEAM paths.
# Source server's NO_STEAM branch omits types still named by local hit recording.
target_compile_options(server PRIVATE -UNO_STEAM)

target_link_libraries(engine PRIVATE "-framework CFNetwork")

target_link_libraries(client PRIVATE SDL2::SDL2)

target_sources(inputsystem PRIVATE "${SOURCE_ROOT}/inputsystem/ios_xkeys.cpp")
target_link_libraries(vguimatsurface PRIVATE "-framework CoreFoundation")

target_sources(client PRIVATE "${PORT_ROOT}/client/text_diagnostics.cpp")

target_link_libraries(engine PRIVATE "-framework CoreFoundation")

target_link_libraries(panorama PRIVATE "-framework CoreFoundation")
